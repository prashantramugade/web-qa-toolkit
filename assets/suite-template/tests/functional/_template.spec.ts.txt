import { test, expect } from '@playwright/test';
import { BasePage } from '../../pages/BasePage';
import { allowWrites } from '../../helpers/site';
import { emailCases, textCases } from '../../helpers/form-data';

// Page object: one class per page or component, holding locators and user actions.
class ContactPage extends BasePage {
  readonly path = '/contact';
  name = this.page.getByLabel('Name');
  email = this.page.getByLabel('Email');
  message = this.page.getByLabel('Message');
  send = this.page.getByRole('button', { name: /send/i });
  status = this.page.getByRole('status');

  async fillAndSend(values: { name: string; email: string; message: string }) {
    await this.name.fill(values.name);
    await this.email.fill(values.email);
    await this.message.fill(values.message);
    await this.send.click();
  }
}

test.describe('Journey: contact the shop', () => {
  test('a valid message is confirmed', async ({ page }) => {
    test.skip(!allowWrites, 'submits data: set ALLOW_WRITES=1 and run against a test environment');
    test.info().annotations.push({ type: 'finding', description: 'F-xxx (optional: link the test to a report finding)' });
    const contact = new ContactPage(page);
    await contact.goto();
    await contact.fillAndSend({ name: 'QA Tester', email: emailCases.valid[0], message: textCases.unicode });
    await expect(contact.status).toContainText(/thank|sent|success/i);
  });
});
